Se puede repetir sin problema. -- Ejecutar en Supabase > SQL Editor. -- ===================================================================== -- 1. Encargado de grupo create table if not exists public.encargados_grupo ( grupo_id integer primary key references public.grupos(id) on delete cascade, docente_id uuid not null references public.perfiles(id) on delete cascade ); alter table public.encargados_grupo enable row level security; revoke all on public.encargados_grupo from anon; drop policy if exists "leer" on public.encargados_grupo; create policy "leer" on public.encargados_grupo for select to authenticated using ((select public.rol_en('administrador','director','coordinador','subcoordinador')) or docente_id = (select auth.uid())); drop policy if exists "escribir" on public.encargados_grupo; create policy "escribir" on public.encargados_grupo for all to authenticated using ((select public.rol_en('administrador','coordinador'))) with check ((select public.rol_en('administrador','coordinador'))); -- Mismas reglas de grado que las asignaciones (coordinador solo sus grados; maestro solo los suyos) drop trigger if exists trg_validar_encargado on public.encargados_grupo; create trigger trg_validar_encargado before insert or update on public.encargados_grupo for each row execute function public.validar_asignacion(); create or replace function public.es_encargado(p_grupo integer) returns boolean language sql stable security definer set search_path = '' as $$ select exists (select 1 from public.encargados_grupo e where e.grupo_id = p_grupo and e.docente_id = (select auth.uid())) $$; -- 2. Actitudinales (asignaturas cuya área empieza con "Hábitos") create or replace function public.puede_ver_actitudinal(p_grupo integer) returns boolean language sql stable security definer set search_path = '' as $$ select public.rol_en('administrador','director','subcoordinador') or (public.rol_en('coordinador') and public.perfil_permite_grado((select auth.uid()), (select g.grado_id from public.grupos g where g.id = p_grupo))) or (public.rol_en('docente') and public.es_encargado(p_grupo)) $$; create or replace function public.puede_editar_actitudinal(p_grupo integer, p_unidad smallint) returns boolean language sql stable security definer set search_path = '' as $$ select public.puede_ver_actitudinal(p_grupo) and public.puede_editar('actitudinales') and public.puede_editar('notas') and ( public.rol_en('administrador','director','coordinador') or public.unidad_abierta_grupo(p_grupo, p_unidad) ) $$; -- Grupos que el usuario puede trabajar (solo grados que tienen hábitos) create or replace function public.grupos_actitudinales() returns table (grupo_id integer, grado text, seccion text, grado_id smallint, etiquetas text[]) language sql stable security definer set search_path = '' as $$ select g.id, gr.nombre, se.nombre, gr.id, gr.etiquetas_periodo from public.grupos g join public.anios_escolares y on y.id = g.anio_id and y.activo join public.grados gr on gr.id = g.grado_id join public.secciones se on se.id = g.seccion_id where public.puede_ver_actitudinal(g.id) and exists (select 1 from public.asignaturas s where s.grado_id = g.grado_id and s.activa and public.norm_txt(s.area) like 'habitos%') order by gr.orden, se.nombre $$; create or replace function public.alumnos_actitudinal(p_grupo integer) returns table (inscripcion_id uuid, apellidos text, nombres text) language plpgsql stable security definer set search_path = '' as $$ begin if not public.puede_ver_actitudinal(p_grupo) then raise exception 'No tiene permiso para este grupo' using errcode = '42501'; end if; return query select i.id, a.apellidos, a.nombres from public.inscripciones i join public.alumnos a on a.id = i.alumno_id where i.grupo_id = p_grupo and a.activo order by a.apellidos, a.nombres; end $$; create or replace function public.actitudinales_grupo(p_grupo integer) returns table (inscripcion_id uuid, asignatura_id integer, unidad smallint, valor_lit text) language plpgsql stable security definer set search_path = '' as $$ begin if not public.puede_ver_actitudinal(p_grupo) then raise exception 'No tiene permiso para este grupo' using errcode = '42501'; end if; return query select n.inscripcion_id, n.asignatura_id, n.unidad, n.valor_lit::text from public.notas n join public.inscripciones i on i.id = n.inscripcion_id and i.grupo_id = p_grupo join public.asignaturas s on s.id = n.asignatura_id where n.valor_lit is not null and public.norm_txt(s.area) like 'habitos%'; end $$; create or replace function public.guardar_actitudinal( p_inscripcion uuid, p_asignatura integer, p_unidad smallint, p_letra text) returns void language plpgsql security definer set search_path = '' as $$ declare v_grupo integer; v_grado smallint; v_letra text := upper(trim(coalesce(p_letra, ''))); begin select i.grupo_id, g.grado_id into v_grupo, v_grado from public.inscripciones i join public.grupos g on g.id = i.grupo_id where i.id = p_inscripcion; if v_grupo is null then raise exception 'Alumno no encontrado'; end if; if not exists (select 1 from public.asignaturas s where s.id = p_asignatura and s.grado_id = v_grado and public.norm_txt(s.area) like 'habitos%') then raise exception 'Esa asignatura no es un actitudinal de este grado'; end if; if p_unidad not between 1 and 4 then raise exception 'Unidad no válida'; end if; if not public.puede_editar_actitudinal(v_grupo, p_unidad) then raise exception 'No puedes cambiar los actitudinales de este grupo o la unidad está cerrada.' using errcode = '42501'; end if; if v_letra = '' then delete from public.notas where inscripcion_id = p_inscripcion and asignatura_id = p_asignatura and unidad = p_unidad; return; end if; if v_letra not in ('A','B','C','D') then raise exception 'La literal debe ser A, B, C o D'; end if; insert into public.notas (inscripcion_id, asignatura_id, unidad, valor_lit, valor_num, actualizado_por) values (p_inscripcion, p_asignatura, p_unidad, v_letra, null, auth.uid()) on conflict (inscripcion_id, asignatura_id, unidad) do update set valor_lit = excluded.valor_lit, valor_num = null, actualizado_por = auth.uid(), actualizado_en = now(); end $$; revoke execute on function public.es_encargado(integer) from public, anon; revoke execute on function public.puede_ver_actitudinal(integer) from public, anon; revoke execute on function public.puede_editar_actitudinal(integer, smallint) from public, anon; revoke execute on function public.grupos_actitudinales() from public, anon; revoke execute on function public.alumnos_actitudinal(integer) from public, anon; revoke execute on function public.actitudinales_grupo(integer) from public, anon; revoke execute on function public.guardar_actitudinal(uuid, integer, smallint, text) from public, anon; grant execute on function public.grupos_actitudinales() to authenticated; grant execute on function public.alumnos_actitudinal(integer) to authenticated; grant execute on function public.actitudinales_grupo(integer) to authenticated; grant execute on function public.guardar_actitudinal(uuid, integer, smallint, text) to authenticated; -- Verificación: debe mostrar 'encargados_grupo' select tablename from pg_tables where schemaname = 'public' and tablename = 'encargados_grupo';
